How the Geometry Solver Works (resolve_layout)When you feed this engine your layout JSON, the engine recursively calculates the exact spatial boundaries. It starts with the full screen canvas (e.g., 1920x1080 at origin 0,0) and splits it down.Here is the recursive solver included in stencil-bsp.rs:impl BspNode {
    /// Recursively resolves the absolute pixel coordinates for every Leaf layout panel.
    pub fn resolve_layout(&self, current_rect: Rect, resolved_panels: &mut Vec<ResolvedPanel>) {
        match self {
            BspNode::Leaf { id, palette, content_type, elements } => {
                resolved_panels.push(ResolvedPanel {
                    id: id.clone(),
                    rect: current_rect,
                    palette: palette.clone(),
                    content_type: content_type.clone(),
                    elements: elements.clone(),
                });
            }
            BspNode::Branch { axis, ratio, child_a, child_b } => {
                let clamped_ratio = ratio.clamp(0.0, 1.0);

                match axis {
                    SplitAxis::Vertical => {
                        let width_a = current_rect.width * clamped_ratio;
                        let width_b = current_rect.width - width_a;

                        let rect_a = Rect { x: current_rect.x, y: current_rect.y, width: width_a, height: current_rect.height };
                        let rect_b = Rect { x: current_rect.x + width_a, y: current_rect.y, width: width_b, height: current_rect.height };

                        child_a.resolve_layout(rect_a, resolved_panels);
                        child_b.resolve_layout(rect_b, resolved_panels);
                    }
                    SplitAxis::Horizontal => {
                        let height_a = current_rect.height * clamped_ratio;
                        let height_b = current_rect.height - height_a;

                        let rect_a = Rect { x: current_rect.x, y: current_rect.y, width: current_rect.width, height: height_a };
                        let rect_b = Rect { x: current_rect.x, y: current_rect.y + height_a, width: current_rect.width, height: height_b };

                        child_a.resolve_layout(rect_a, resolved_panels);
                        child_b.resolve_layout(rect_b, resolved_panels);
                    }
                }
            }
        }
    }
}

struct PanelUniforms {
    tint_color: vec4<f32>,
    refraction_index: f32, // Set to 1.52 for cyber-arcane glassmorphism
    opacity: f32,
    blur_radius: f32,      // Set to 20.0px for panel optics
};

@group(0) @binding(0) var<uniform> uniforms: PanelUniforms;
@group(0) @binding(1) var texture_sampler: sampler;

// Texture 1: The actual visual content of THIS layer (rendered elements)
@group(0) @binding(2) var layer_texture: texture_2d<f32>;

// Texture 2: The aggregated texture of everything rendered BEHIND this panel
// (used to compute refraction and background blur)
@group(0) @binding(3) var background_texture: texture_2d<f32>;

struct VertexOutput {
    @builtin(position) position: vec4<f32>,
    @location(0) uv: vec2<f32>,
};

@fragment
fn fs_main(in: VertexOutput) -> @location(0) vec4<f32> {
    // 1. Calculate refraction offset
    // Simulated offset based on the panel's refractive index (e.g., 1.52) [1]
    let offset_direction = vec2<f32>(0.005, -0.005);
    let refracted_uv = in.uv + (offset_direction * (uniforms.refraction_index - 1.0));
    
    // 2. Sample the background texture with the refractive distortion
    let back_color = textureSample(background_texture, texture_sampler, refracted_uv);
    
    // 3. Sample the UI panel's native drawn contents (the "foreground" of this leaf)
    let panel_content = textureSample(layer_texture, texture_sampler, in.uv);
    
    // 4. Mix background content with our glass tint
    let glass_base = mix(back_color, uniforms.tint_color, uniforms.tint_color.a);
    
    // 5. Composite the panel content on top using its alpha layer
    let final_color = mix(glass_base, panel_content, panel_content.a);
    
    // 6. Apply global layer opacity for smooth fading animations [2, 3]
    return vec4<f32>(final_color.rgb, final_color.a * uniforms.opacity);
}

The engine handles this dynamically:/// Converts absolute screen coordinates to local, infinite canvas coordinates.
pub fn screen_to_canvas(&self, screen_pos: egui::Pos2, viewport_rect: egui::Rect) -> egui::Pos2 {
    let zoom = self.primary_canvas_state.zoom;
    let pan = self.primary_canvas_state.pan;
    let relative_pos = screen_pos - viewport_rect.min;
    egui::pos2(
        (relative_pos.x - pan.x) / zoom,
        (relative_pos.y - pan.y) / zoom,
    )
}

/// Translates a 2D screen mouse drag into a 3D rotation Quaternion.
/// This acts as a virtual Arcball rotation widget.
pub fn calculate_gantry_rotation(
    drag_delta: Vec2,
    viewport_size: Vec2,
    sensitivity: f32,
) -> Quat {
    // 1. Convert the screen delta to normalized coordinates (-1.0 to 1.0)
    let dx = (drag_delta.x / viewport_size.x) * sensitivity;
    let dy = (drag_delta.y / viewport_size.y) * sensitivity;

    // 2. Map X drag to rotation around the 3D Y-axis (Yaw)
    let yaw = Quat::from_rotation_y(-dx);
    // 3. Map Y drag to rotation around the 3D X-axis (Pitch)
    let pitch = Quat::from_rotation_x(-dy);

    // 4. Combine them to get the final local rotation change
    yaw * pitch
}

/// Translates a 2D screen mouse drag on an orthographic plane into a 3D translation offset.
pub fn calculate_gantry_translation(
    drag_delta: Vec2,
    view_mode: OrthographicView,
    zoom: f32,
) -> Vec3 {
    let world_dx = drag_delta.x / zoom;
    let world_dy = -drag_delta.y / zoom; // Invert Y because screen space flows down

    match view_mode {
        OrthographicView::Front => Vec3::new(world_dx, world_dy, 0.0),
        OrthographicView::Side  => Vec3::new(0.0, world_dy, world_dx), // Maps to Y and Z
        OrthographicView::Top   => Vec3::new(world_dx, 0.0, world_dy), // Maps to X and Z
    }
}

pub enum OrthographicView {
    Front,
    Side,
    Top,

pub fn project_3d_to_screen(
    pos_3d: Vec3,
    view_proj: Mat4,
    viewport_rect: egui::Rect,
) -> Option<egui::Pos2> {
    let clip_space = view_proj.project_point3(pos_3d);
    
    // Check if the point is behind the camera
    if clip_space.z < -1.0 || clip_space.z > 1.0 {
        return None;
    }

    // Convert normalized device coordinates (-1.0 to 1.0) to screen pixels
    let x = viewport_rect.min.x + (clip_space.x + 1.0) * 0.5 * viewport_rect.width();
    let y = viewport_rect.min.y + (1.0 - clip_space.y) * 0.5 * viewport_rect.height(); // Invert Y

    Some(egui::pos2(x, y))
}
